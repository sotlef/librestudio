# oox (module oox) — the OOXML (OOXML/MS-* binary) import/export filter core:
# DrawingML/VML shape and chart handling, the fast-parser/context-handler XML
# core, OOXML document-properties and crypto (agile/standard encryption),
# PPT/PPTX slide and animation import, OLE/VBA project handling, and the
# oox::core::FastTokenHandler/token maps other modules' OOXML filters build
# on. Source list from Library_oox.mk. The WASM_CHART guard
# (ENABLE_WASM_STRIP_CHART) is unset in this build, so the chart export/import
# sources are included unconditionally; the OS-conditional
# drawingml/color.cxx compile-options split (iOS gets -O0 to dodge an Xcode
# 5.1.1 codegen bug) is Linux-only here, so color.cxx is just a normal
# exception object. Not ported: the GCC/IA64 -Os workaround and the M68K
# -mlong-jump-table-offsets cxxflag, neither of which applies to this build.
#
# gb_Library_use_api(oox, oovbaapi) pulls in the VBA-compatibility API headers
# (oovbaapi/CMakeLists.txt) alongside the normal udkapi/offapi set -- the same
# way basic/CMakeLists.txt's sb library already needs them.
#
# gb_Library_use_custom_headers(oox, oox/generated) is CustomTarget_generated.mk:
# oox/source/token/{tokens,namespaces,properties}.txt are each run through a
# generator script (solenv/bin/generate-tokens.py for tokens.txt itself;
# namespaces.py and properties.py, two oox-specific variants, for the other
# two) to produce headers oox/source/token/tokenmap.cxx, namespacemap.cxx and
# propertynames.cxx include, plus a gperf source that becomes the perfect-hash
# table oox/source/core/fasttokenhandler.cxx needs. This is the same
# generate-tokens.py/gperf pipeline xmloff/CMakeLists.txt already set up for
# xmloff's own tokens.txt (and that svx/CMakeLists.txt stubbed out for oox's
# tokens.hxx only, back when oox itself had not been ported yet) -- here it is
# reproduced in full, including the namespaces/properties variants and the
# tokenhash.inc gperf stage. CustomTarget_generated.mk's other two outputs,
# misc/oox-drawingml-adj-names and misc/vml-shape-types (generated by
# oox/source/export/preset-definitions-to-shape-types.pl and packaged by
# Package_generated.mk to $BRAND_BASE_DIR/share/filter), are read by
# drawingml.cxx/vmlexport.cxx/fontworkhelpers.cxx only at *runtime*
# (OUString paths under $BRAND_BASE_DIR) -- not compiled into the library --
# so they are out of scope for building the oox target itself.

find_program(PYTHON3_EXECUTABLE NAMES python3 REQUIRED)
find_program(GPERF_EXECUTABLE gperf REQUIRED)

set(OOX_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated" CACHE INTERNAL
    "oox generated token/namespace/property headers directory")
set(OOX_TOKEN_SRC_DIR "${CMAKE_SOURCE_DIR}/oox/source/token")

# --- tokens.hxx / tokenhash.inc (generate-tokens.py + gperf) ------------------

set(OOX_TOKENS_HXX "${OOX_GENERATED_DIR}/oox/token/tokens.hxx")
add_custom_command(
    OUTPUT "${OOX_TOKENS_HXX}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OOX_GENERATED_DIR}/misc"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OOX_GENERATED_DIR}/oox/token"
    COMMAND "${PYTHON3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/solenv/bin/generate-tokens.py"
            "${OOX_TOKEN_SRC_DIR}/tokens.txt"
            "${OOX_GENERATED_DIR}/misc/tokenids.inc"
            "${OOX_GENERATED_DIR}/tokennames.inc"
            "${OOX_GENERATED_DIR}/misc/tokenhash.gperf"
    COMMAND ${CMAKE_COMMAND} -E cat
            "${OOX_TOKEN_SRC_DIR}/tokens.hxx.head"
            "${OOX_GENERATED_DIR}/misc/tokenids.inc"
            "${OOX_TOKEN_SRC_DIR}/tokens.hxx.tail" > "${OOX_TOKENS_HXX}"
    DEPENDS "${OOX_TOKEN_SRC_DIR}/tokens.txt"
            "${OOX_TOKEN_SRC_DIR}/tokens.hxx.head"
            "${OOX_TOKEN_SRC_DIR}/tokens.hxx.tail"
            "${CMAKE_SOURCE_DIR}/solenv/bin/generate-tokens.py"
    COMMENT "generate-tokens.py: oox/token/tokens.hxx"
    VERBATIM)

# Same gperf/sed fixup as xmloff/CMakeLists.txt's tokenhash.inc.
set(OOX_TOKENHASH_INC "${OOX_GENERATED_DIR}/tokenhash.inc")
add_custom_command(
    OUTPUT "${OOX_TOKENHASH_INC}"
    COMMAND sh -c "'${GPERF_EXECUTABLE}' --compare-strncmp --switch=2 --readonly-tables --pic '${OOX_GENERATED_DIR}/misc/tokenhash.gperf' | sed -e '/^#line/d' -e 's/(char\\*)0/(char\\*)0, 0/g' > '${OOX_TOKENHASH_INC}'"
    DEPENDS "${OOX_TOKENS_HXX}"
    COMMENT "gperf: oox/generated/tokenhash.inc"
    VERBATIM)

# --- namespaces.hxx / namespacenames.inc / namespaces-strictnames.inc --------

set(OOX_NAMESPACES_HXX "${OOX_GENERATED_DIR}/oox/token/namespaces.hxx")
set(OOX_NAMESPACENAMES_INC "${OOX_GENERATED_DIR}/namespacenames.inc")
set(OOX_NAMESPACES_STRICTNAMES_INC "${OOX_GENERATED_DIR}/namespaces-strictnames.inc")
add_custom_command(
    OUTPUT "${OOX_NAMESPACES_HXX}" "${OOX_NAMESPACENAMES_INC}" "${OOX_NAMESPACES_STRICTNAMES_INC}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OOX_GENERATED_DIR}/misc"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OOX_GENERATED_DIR}/oox/token"
    COMMAND "${PYTHON3_EXECUTABLE}" "${OOX_TOKEN_SRC_DIR}/namespaces.py"
            "${OOX_TOKEN_SRC_DIR}/namespaces.txt"
            "${OOX_GENERATED_DIR}/misc/namespaceids.inc"
            "${OOX_NAMESPACENAMES_INC}"
            "${OOX_GENERATED_DIR}/misc/namespaces.txt"
            "${OOX_TOKEN_SRC_DIR}/namespaces-strict.txt"
            "${OOX_NAMESPACES_STRICTNAMES_INC}"
    COMMAND ${CMAKE_COMMAND} -E cat
            "${OOX_TOKEN_SRC_DIR}/namespaces.hxx.head"
            "${OOX_GENERATED_DIR}/misc/namespaceids.inc"
            "${OOX_TOKEN_SRC_DIR}/namespaces.hxx.tail" > "${OOX_NAMESPACES_HXX}"
    DEPENDS "${OOX_TOKEN_SRC_DIR}/namespaces.txt"
            "${OOX_TOKEN_SRC_DIR}/namespaces-strict.txt"
            "${OOX_TOKEN_SRC_DIR}/namespaces.hxx.head"
            "${OOX_TOKEN_SRC_DIR}/namespaces.hxx.tail"
            "${OOX_TOKEN_SRC_DIR}/namespaces.py"
    COMMENT "namespaces.py: oox/token/namespaces.hxx"
    VERBATIM)

# --- properties.hxx / propertynames.inc ---------------------------------------

set(OOX_PROPERTIES_HXX "${OOX_GENERATED_DIR}/oox/token/properties.hxx")
set(OOX_PROPERTYNAMES_INC "${OOX_GENERATED_DIR}/propertynames.inc")
add_custom_command(
    OUTPUT "${OOX_PROPERTIES_HXX}" "${OOX_PROPERTYNAMES_INC}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OOX_GENERATED_DIR}/misc"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OOX_GENERATED_DIR}/oox/token"
    COMMAND "${PYTHON3_EXECUTABLE}" "${OOX_TOKEN_SRC_DIR}/properties.py"
            "${OOX_TOKEN_SRC_DIR}/properties.txt"
            "${OOX_GENERATED_DIR}/misc/propertyids.inc"
            "${OOX_PROPERTYNAMES_INC}"
    COMMAND ${CMAKE_COMMAND} -E cat
            "${OOX_TOKEN_SRC_DIR}/properties.hxx.head"
            "${OOX_GENERATED_DIR}/misc/propertyids.inc"
            "${OOX_TOKEN_SRC_DIR}/properties.hxx.tail" > "${OOX_PROPERTIES_HXX}"
    DEPENDS "${OOX_TOKEN_SRC_DIR}/properties.txt"
            "${OOX_TOKEN_SRC_DIR}/properties.hxx.head"
            "${OOX_TOKEN_SRC_DIR}/properties.hxx.tail"
            "${OOX_TOKEN_SRC_DIR}/properties.py"
    COMMENT "properties.py: oox/token/properties.hxx"
    VERBATIM)

add_custom_target(oox_generated DEPENDS
    "${OOX_TOKENS_HXX}" "${OOX_TOKENHASH_INC}"
    "${OOX_NAMESPACES_HXX}" "${OOX_NAMESPACENAMES_INC}" "${OOX_NAMESPACES_STRICTNAMES_INC}"
    "${OOX_PROPERTIES_HXX}" "${OOX_PROPERTYNAMES_INC}")

# gb_Helper_optional(AVMEDIA, avmedia) in Library_oox.mk:41. ENABLE_AVMEDIA
# defaults ON (cmake/LibreStudioConfigHeaders.cmake, see avmedia/
# CMakeLists.txt's own comment for the configure.ac evidence) -- same

    target_link_libraries(oox PRIVATE avmedia)
endif ()

# The crypto backend is chosen by LIBRESTUDIO_TLS_BACKEND, the same switch
# comphelper's and svl's crypto are built with; oox/source/crypto links
# whatever that backend is (gb_Library_use_externals,oox,openssl/openssl_headers
# under TLS=OPENSSL, plc4/nss3 under TLS=NSS).
if (LIBRESTUDIO_TLS_BACKEND STREQUAL "NSS")
    pkg_check_modules(NSS REQUIRED IMPORTED_TARGET nss)
    target_link_libraries(oox PRIVATE PkgConfig::NSS)
else ()
    pkg_check_modules(OPENSSL REQUIRED IMPORTED_TARGET openssl)
    target_link_libraries(oox PRIVATE PkgConfig::OPENSSL)
endif ()
