# test — C++ CppunitTest fixture helpers one tier up from unotest: VCL/office
# bootstrap (BootstrapFixture, screenshot tests), XML/HTML diffing, and a
# handful of UNO API test helpers (form/shape/transferable).  Source lists
# from Library_test_setupvcl.mk and Library_test.mk.  Same relationship to
# sc/sd/sw/... as unotest (see unotest/CMakeLists.txt): those modules only
# need this for their own CppunitTest targets, not for their production
# libraries.
#
# Out of scope for this tier, and why:
#   - CppunitTest_test_a11y.mk / CppunitTest_test_xpath.mk: tests of this
#     module itself, not something anything else depends on.
#
# subsequenttest (Library_subsequenttest.mk) IS ported below, in its own
# section right after test_fixtures: ~180 UNO API interface-test helpers
# (test/source/sheet/*, .../text/*, .../a11y/*, .../table/*, ...),
# historically used only by subsequentcheck (API-level tests against a
# running office), but also needed by officecfg_accelerator_key_checker_test,
# toolkit_a11y and vcl_a11y -- e.g. officecfg/qa/acceleratorkeychecker.cxx
# #includes <test/unoapi_test.hxx>, which is compiled into subsequenttest
# (Library_subsequenttest.mk line 47's test/source/unoapi_test), not into
# test_fixtures above. Now that subsequenttest is a real target,
# officecfg_accelerator_key_checker_test and toolkit_a11y are wired (see
# officecfg/CppunitTest_deferred.cmake and toolkit/CppunitTest_deferred.cmake
# respectively); vcl_a11y is not, still a separate follow-up task.
#
# Package_unittest.mk IS now needed, by librestudio_add_cppunit_test()'s
# CONFIGURATION option (cmake/LibreStudioTest.cmake -- read its header
# comment first for the full mechanism): it packages test/user-template/
# (registry-common/modifications.xcd, registry-user-ui/user-ui.xcd, plus a
# full synthetic user profile -- user/database, user/wordbook, user/autotext,
# user/autocorr, user/config/soffice.cfg/...) into $(WORKDIR)/unittest, which
# both feeds gb_CppunitTest_use_configuration's two non-instdir
# CONFIGURATION_LAYERS entries AND is what gb_CppunitTest__make_args's
# generic "cp -r $(WORKDIR)/unittest $@.user" step copies into every use_ure
# test's .user directory. Staged here, once, as a plain file(COPY ...) at
# configure time -- test/user-template is static source content, not
# generated by any build step, so there is nothing a custom command would add
# over copying it directly (same simplification officecfg/CMakeLists.txt's
# own main.list/main.sed writes already accept: reconfigure to pick up an
# edit to test/user-template's contents). LIBRESTUDIO_TEST_UNITTEST_DIR is a
# CACHE INTERNAL variable (not a plain one) specifically so it stays visible
# to librestudio_add_cppunit_test() calls made later from other directories'
# own CMakeLists.txt, and from the deferred *_CppunitTest_deferred.cmake
# files include()'d from the root CMakeLists.txt's own scope -- the same
# visibility need OFFICECFG_INCLUDE_DIR/LIBRESTUDIO_REGISTRY_DIR already have
# for the same reason.
set(LIBRESTUDIO_TEST_UNITTEST_DIR "${CMAKE_BINARY_DIR}/test/unittest" CACHE INTERNAL
    "staged copy of test/user-template, mirroring gbuild's \$(WORKDIR)/unittest")
file(COPY "${CMAKE_SOURCE_DIR}/test/user-template/"
     DESTINATION "${LIBRESTUDIO_TEST_UNITTEST_DIR}")

# --- vclbootstrapprotector ---------------------------------------------------

# cppunit's headers are used inline by vclbootstrapprotector.cxx below; this
# is the first target in this directory scope that needs them (the "test"
# section further down used to be first and re-called this for the same
# directory-scope reason unotest/CMakeLists.txt documents -- now guarded
# there instead of duplicated, since this call already covers the whole
# file).
pkg_check_modules(CPPUNIT REQUIRED IMPORTED_TARGET cppunit)

# cppunit "protector" plug-in (Library_vclbootstrapprotector.mk, 1 source
# file), used by gb_CppunitTest_use_vcl.  Same category as unotest's
# unobootstrapprotector/unoexceptionprotector (see unotest/CMakeLists.txt for
# the full reasoning this mirrors): never linked against by anything in the
# tree -- CppunitTest.mk's gb_CppunitTest_use_vcl passes it to cppunittester
# as "--protector <path> <symbol>", which dlopens it via
# osl::Module::getFunctionSymbol (CppUnit::PlugInManager), the same loading
# model librestudio_add_cppunit_test()'s own PROTECTORS argument implements.
# So this is a CMake MODULE library, not SHARED, and gets no install() rule.
#
# Output naming: Repository.mk registers vclbootstrapprotector under the same
# PLAINLIBS_NONE category as test-setupvcl/unobootstrapprotector/
# unoexceptionprotector/unotest/test (all fixed this session from a wrong OOO
# to the correct PLAIN) -- bare "libvclbootstrapprotector.so", no "lo" suffix
# (solenv/gbuild/platform/unxgcc.mk's gb_Library_FILENAMES).  PLAINLIBS_NONE
# is also in gb_Library_VALIDGROUPS_NOTINSTALLED (solenv/gbuild/Helper.mk),
# so upstream never installs this library either.

set(VCLBOOTSTRAPPROTECTOR_SOURCES
    test/source/vclbootstrapprotector
)
librestudio_expand_cxx(VCLBOOTSTRAPPROTECTOR_SOURCES)
add_library(vclbootstrapprotector MODULE ${VCLBOOTSTRAPPROTECTOR_SOURCES})

target_include_directories(vclbootstrapprotector PRIVATE
    "${udkapi_normal_HEADERS_INCLUDE_DIR}"
    "${offapi_normal_HEADERS_INCLUDE_DIR}")
add_dependencies(vclbootstrapprotector udkapi_normal_headers offapi_normal_headers)

target_link_libraries(vclbootstrapprotector PRIVATE
    comphelper cppu i18nlangtag sal test-setupvcl tl utl vcl
    PkgConfig::CPPUNIT
    Boost::boost)

librestudio_set_library_output_name(vclbootstrapprotector vclbootstrapprotector PLAIN)
